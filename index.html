Google Sheets Integration
Conecta el dashboard a tu Google Sheet vía Apps Script.

URL de Apps Script configurada. Tu dashboard está conectado en modo bidireccional: envía datos a Sheets y también los carga de vuelta.

📋 Código completo de Apps Script

Borra todo lo que tengas en Apps Script y reemplaza con este código. Luego deployar como Web App → Anyone.

// ══════════════════════════════════════════════
//  Viviana · Life OS — Apps Script Backend
// ══════════════════════════════════════════════

function doPost(e) {
  try {
    // Lee del form submit (e.parameter.data) o fetch directo
    const raw = (e.parameter && e.parameter.data)
                  ? e.parameter.data
                  : e.postData.contents;
    const data = JSON.parse(raw);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (data.type === 'finanzas') {
      let sh = ss.getSheetByName('Finanzas');
      if (!sh) {
        sh = ss.insertSheet('Finanzas');
        sh.appendRow(['date','type','debtName','debtIndex',
                      'monto','nota','ts','daily_total','ant_total']);
      }
      if (data.debt_payments && data.debt_payments.length > 0) {
        data.debt_payments.forEach(p => {
          sh.appendRow([p.date,'deuda',p.debtName,p.debtIndex,
                        p.monto,p.nota||'',p.ts||'','','']);
        });
      }
      sh.appendRow([data.date,'resumen_diario','','','','',
                    Date.now(),data.daily_total||0,data.ant_total||0]);
    } else {
      let sh = ss.getSheetByName('Life OS');
      if (!sh) {
        sh = ss.insertSheet('Life OS');
        sh.appendRow(['date','completion','pillars',
                      'win1','win2','win3','work1','work2','work3',
                      'jobs','applications','contacts','interviews',
                      'review1','review2','review3']);
      }
      sh.appendRow([
        data.date, data.completion,
        Array.isArray(data.pillars) ? data.pillars.join(',') : (data.pillars||''),
        data.win1||'', data.win2||'', data.win3||'',
        data.work1||'', data.work2||'', data.work3||'',
        data.jobs||'', data.applications||'', data.contacts||'', data.interviews||'',
        data.review1||'', data.review2||'', data.review3||''
      ]);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const result = {};

  try {
    const sh = ss.getSheetByName('Life OS');
    if (sh && sh.getLastRow() > 1) {
      const hdr = sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0];
      const row = sh.getRange(sh.getLastRow(),1,1,sh.getLastColumn()).getValues()[0];
      const obj = {};
      hdr.forEach((h,i) => { if(h) obj[h] = row[i]; });
      result.lifeOS = obj;
    }
  } catch(ex) { result.lifeOS = null; }

  try {
    const shF = ss.getSheetByName('Finanzas');
    if (shF && shF.getLastRow() > 1) {
      const hdr = shF.getRange(1,1,1,shF.getLastColumn()).getValues()[0];
      const rows = shF.getRange(2,1,shF.getLastRow()-1,shF.getLastColumn()).getValues();
      result.finanzas = rows.map(row => {
        const obj = {};
        hdr.forEach((h,i) => { if(h) obj[h] = row[i]; });
        return obj;
      });
    }
  } catch(ex) { result.finanzas = []; }

  // JSONP — necesario para leer la respuesta desde el browser
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) {
    return ContentService
      .createTextOutput(cb + '(' + JSON.stringify(result) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
